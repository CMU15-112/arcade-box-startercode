  \tikz[baseline=-0.75ex]{%
    \node[circle, fill=#1, draw=#1!55!black, line width=1.5pt,
          text=white, font=\bfseries\normalsize, minimum size=1.2em,
          inner sep=0pt] {#2};}}

{\huge\color{arcdark}\bfseries Please follow these simple rules when using me:}
\vspace{3mm}

\LARGE\raggedright
\begin{enumerate}[leftmargin=2.2em, labelsep=0.6em,
                  itemsep=3mm, topsep=0pt]
  \item Most games can be quit by pressing the \Pone\ button.
  \item If a game is stuck and won't quit, press \Reset\ \textbf{once} and wait.
        The game will close and the menu will come back in a few seconds.
        If it still doesn't come back, hold \Reset\ for 5~seconds to restart me.
  \item 15-112 students who want to port their games to this box should
        contact Prof.~Riley.
  \item Please don't connect or disconnect things from me, including USB devices
        like keyboards. If you have debugging issues, please contact Prof.~Riley.
  \item Please don't break me or load other software onto me.
\end{enumerate}
